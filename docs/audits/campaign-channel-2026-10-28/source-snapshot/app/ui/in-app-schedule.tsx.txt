function Select({ label, value, options, onChange, placeholder, t, searchable = false, invalid = false, disabled = false }: { label: string; value: string; options: Option[]; onChange: (value: string) => void; placeholder?: string; t: Text; searchable?: boolean; invalid?: boolean; disabled?: boolean }) {
  const id = useId(), root = useRef<HTMLDivElement>(null), button = useRef<HTMLButtonElement>(null), input = useRef<HTMLInputElement>(null);
  const [open, setOpen] = useState(false), [query, setQuery] = useState(""), [active, setActive] = useState(0);
  const filtered = options.filter(option => option.label.toLowerCase().includes(query.toLowerCase()));
  const selected = options.find(option => option.value === value);
  const close = () => { setOpen(false); setQuery(""); };
  useEffect(() => { if (!open) return; const listener = (e: MouseEvent) => { if (!root.current?.contains(e.target as Node)) close(); }; document.addEventListener("mousedown", listener); if (searchable) input.current?.focus(); return () => document.removeEventListener("mousedown", listener); }, [open, searchable]);
  const choose = (next: string) => { onChange(next); close(); button.current?.focus(); };
  const keyDown = (e: React.KeyboardEvent) => {
    if (e.key === "Escape") { e.preventDefault(); e.stopPropagation(); close(); button.current?.focus(); }
    if (["ArrowDown", "ArrowUp", "Home", "End"].includes(e.key)) { e.preventDefault(); if (!open) { setOpen(true); setActive(0); } else setActive(index => e.key === "Home" ? 0 : e.key === "End" ? filtered.length - 1 : Math.max(0, Math.min(filtered.length - 1, index + (e.key === "ArrowDown" ? 1 : -1)))); }
    if (e.key === "Enter" && open) { e.preventDefault(); if (filtered[active]) choose(filtered[active].value); }
    if (e.key === "Tab") close();
  };
  return <div className={styles.select} ref={root} onKeyDown={keyDown}>
    <button type="button" ref={button} role="combobox" aria-label={t(label)} aria-expanded={open} aria-controls={`${id}-options`} aria-haspopup="listbox" aria-invalid={invalid || undefined} aria-activedescendant={open && filtered[active] ? `${id}-${active}` : undefined} disabled={disabled} className={`${styles.selectControl} ${invalid ? styles.invalid : ""}`} onClick={() => { setOpen(!open); setQuery(""); setActive(Math.max(0, options.findIndex(option => option.value === value))); }}><span className={!selected ? styles.placeholder : ""}>{selected?.label ?? (value || t(placeholder ?? "Select..."))}</span><span className={styles.selectArrow}><ChevronDown size={17}/></span></button>
    {open && <div className={styles.selectMenu}>{searchable && <div className={styles.selectSearch}><Search size={14}/><input ref={input} aria-label={`${t("Search")} ${t(label)}`} value={query} placeholder={t("Search...")} onChange={e => { setQuery(e.target.value); setActive(0); }}/></div>}<div id={`${id}-options`} role="listbox" aria-label={t(label)} className={styles.optionList}>{filtered.length ? filtered.map((option, index) => <button key={option.value} id={`${id}-${index}`} type="button" role="option" aria-selected={value === option.value} className={index === active ? styles.activeOption : ""} onMouseEnter={() => setActive(index)} onClick={() => choose(option.value)}>{option.label}</button>) : <div className={styles.noResults}>{t("No results found")}</div>}</div></div>}
  </div>;
}

        {triggers.map((item, index) => <div className={styles.trigger} key={item.id} data-trigger-id={item.id}>{index > 0 && <span className={styles.triggerOr}>{t("OR")}</span>}<div className={styles.triggerFields}><b className={styles.triggerTitle}>{t(item.type)}</b>
          {item.type === "Start Session" ? <><label className={styles.radio}><input type="radio" name={`session-${item.id}`} checked={!item.specificApp} onChange={() => changeTrigger(item.id, { specificApp: false })}/>{t("Start session in any app")}</label><label className={styles.radio}><input type="radio" name={`session-${item.id}`} checked={Boolean(item.specificApp)} onChange={() => changeTrigger(item.id, { specificApp: true })}/>{t("Start session in a specific app")}</label>{item.specificApp && <Select label="Choose an app" t={t} value={item.appId ?? ""} options={metadata.apps.map(row => ({ value: row.id, label: row.name }))} placeholder="Choose an app" searchable onChange={appId => changeTrigger(item.id, { appId })}/>}</> : (item.type === "Perform Custom Event" || ecommerceEvents[item.type]) ? <label className={styles.field}><b>{t("Custom Event Name")}</b><Select label="Choose a custom event" t={t} value={item.eventName ?? ""} placeholder={loading ? "Loading..." : "Choose a custom event"} options={item.eventName && !eventOptions.some(option => option.value === item.eventName) ? [...eventOptions, { value: item.eventName, label: item.eventName }] : eventOptions} searchable disabled={loading} onChange={eventName => changeTrigger(item.id, { eventName, filtersEnabled: false, filterGroups: [] })}/></label> : item.type === "Interact With Campaign" ? <><label className={styles.radio}><input type="radio" name={`interaction-${item.id}`} checked={!item.specificCampaign} onChange={() => changeTrigger(item.id, { specificCampaign: false })}/>{t("Open Push From Any Campaign")}</label><label className={styles.radio}><input type="radio" name={`interaction-${item.id}`} checked={Boolean(item.specificCampaign)} onChange={() => changeTrigger(item.id, { specificCampaign: true })}/>{t("Open Push From A Specific Campaign")}</label>{item.specificCampaign && <Select label="Choose a campaign" t={t} value={item.campaignId ?? ""} placeholder="Choose a campaign" searchable options={metadata.campaigns.filter(row => row.id !== draft.id && row.channel === "push").map(row => ({ value: row.id, label: row.name }))} onChange={campaignId => changeTrigger(item.id, { campaignId })}/>}</> : null}
          {item.type !== "Start Session" && item.type !== "Interact With Campaign" && <><label className={styles.checkbox}><input type="checkbox" checked={Boolean(item.filtersEnabled)} onChange={e => changeTrigger(item.id, { filtersEnabled: e.target.checked, filterGroups: item.filterGroups?.length ? item.filterGroups : [newGroup()] })}/>{t("Add property filters")}</label>{item.filtersEnabled && <PropertyEditor trigger={item} properties={metadata.events.find(event => event.name === item.eventName)?.properties ?? {}} t={t} update={patch => changeTrigger(item.id, patch)}/>}</>}
        </div><button type="button" className={styles.removeTrigger} aria-label={`${t("Close")} ${t(item.type)}`} onClick={() => change({ iamTriggers: triggers.filter(row => row.id !== item.id) })}><X size={20}/></button></div>)}
        {!triggers.length && <p className={styles.empty}>{t("No triggers. Select a trigger below to continue.")}</p>}
        <div className={styles.triggerAdd}><p>{t("Evaluate if user takes specified actions")}</p><div><Select label="Select Trigger" t={t} value={trigger} placeholder="Select Trigger" options={inAppTriggerTypes.map(value => ({ value, label: t(value) }))} onChange={setTrigger}/><button type="button" className={styles.addTrigger} disabled={!trigger} onClick={() => { change({ iamTriggers: [...triggers, { id: crypto.randomUUID(), type: trigger as InAppTrigger["type"], ...(ecommerceEvents[trigger] ? { eventName: ecommerceEvents[trigger] } : {}) }] }); setTrigger(""); }}><Plus size={19}/>{t("Add Trigger")}</button></div></div>
      </div></div>
      <div className={styles.section}><h3>{t("Schedule Delay")}</h3><p>{t("Choose when to send the campaign after the trigger event occurs.")}</p><div className={styles.field}><b>{t("Send")}</b></div><div className={styles.delayFields}><Select label="Send" t={t} value={delay.mode} options={[{ value: "immediately", label: t("Immediately") }, { value: "delay", label: t("After a delay") }]} onChange={mode => setDelay({ mode })}/>{delay.mode === "delay" && <><input type="number" aria-label={t("Delay")} min={0} step={1} value={delay.value} onChange={e => setDelay({ value: Number(e.target.value) })}/><Select label="Time unit for schedule delay" t={t} value={delay.unit} options={["seconds", "minutes", "hours"].map(value => ({ value, label: t(value[0].toUpperCase() + value.slice(1)) }))} onChange={unit => setDelay({ unit })}/></>}</div></div>
      <div className={styles.section}><h3>{t("Campaign duration")}</h3><div className={styles.field}><b>{t("Start Time (Required)")}</b></div><div className={styles.dateFields}><input type="date" aria-label={t("Select Date")} value={startDate} required onInput={e => change({ startDate: e.currentTarget.value })} onChange={e => change({ startDate: e.target.value })}/><input type="time" aria-label={t("Start time")} value={startTime} required onInput={e => change({ sendTime: e.currentTarget.value })} onChange={e => change({ sendTime: e.target.value })}/><Select label="Time zone" t={t} value={String(delivery.timezone ?? "Company time")} options={[{ value: "Company time", label: t("Company time - Eastern Time (US & Canada)") }, { value: "Recipient local time", label: t("Recipient local time") }, { value: "UTC", label: "UTC" }]} onChange={timezone => change({ timezone })}/></div><label className={styles.checkbox}><input type="checkbox" checked={Boolean(delivery.endEnabled)} onChange={e => change({ endEnabled: e.target.checked })}/>{t("End Time (Optional)")}</label>{Boolean(delivery.endEnabled) && <div className={styles.dateFields}><input type="date" aria-label={t("End date")} value={endTime.split("T")[0] ?? ""} min={startDate} onInput={e => change({ endTime: `${e.currentTarget.value}T${endTime.split("T")[1] || startTime}` })} onChange={e => change({ endTime: `${e.target.value}T${endTime.split("T")[1] || startTime}` })}/><input type="time" aria-label={t("End time")} value={endTime.split("T")[1] || startTime} onInput={e => change({ endTime: `${endTime.split("T")[0] || startDate}T${e.currentTarget.value}` })} onChange={e => change({ endTime: `${endTime.split("T")[0] || startDate}T${e.target.value}` })}/></div>}</div>
      <ul className={styles.summary}><li>{t("Schedule delay:")} <b>{delay.mode === "immediately" ? t("immediately after trigger criteria are met") : `${delay.value} ${t(delay.unit[0].toUpperCase() + delay.unit.slice(1))} ${t("after trigger criteria are met")}`}</b></li>{startDate && <li>{t("Campaign start:")} <b>{formatDate(`${startDate}T${startTime}`)}</b></li>}{Boolean(delivery.endEnabled) && endTime && <li>{t("Campaign end:")} <b>{formatDate(endTime)}</b></li>}</ul>
      {errors.length > 0 && <ul className={styles.validation} aria-live="polite">{errors.map(issue => <li key={issue}>{t(issue)}</li>)}</ul>}
    </section>
    <section className={styles.panel}><h2 className={styles.heading}>{t("Delivery Controls")}</h2><p>{t("Manage the number of messages your users will receive.")}</p><label className={styles.checkbox}><input type="checkbox" checked={Boolean(delivery.reeligible)} onChange={e => change({ reeligible: e.target.checked })}/>{t("Allow users to become re-eligible to receive campaign")}</label>{Boolean(delivery.reeligible) && <div className={styles.reentry}><b>{t("Re-eligibility after")}</b><input aria-label={t("Re-eligibility after")} type="number" min={1} value={Number(delivery.reentryValue ?? delivery.reentryDays ?? 7)} onChange={e => { const value = Number(e.target.value); const unit = String(delivery.reentryUnit ?? "days"); change({ reentryValue: value, reentryUnit: unit, reentryDays: unit === "hours" ? value / 24 : unit === "weeks" ? value * 7 : value }); }}/><Select label="Time unit for re-eligibility" t={t} value={String(delivery.reentryUnit ?? "days")} options={["hours", "days", "weeks"].map(value => ({ value, label: t(value[0].toUpperCase() + value.slice(1)) }))} onChange={unit => { const value = Number(delivery.reentryValue ?? delivery.reentryDays ?? 7); change({ reentryValue: value, reentryUnit: unit, reentryDays: unit === "hours" ? value / 24 : unit === "weeks" ? value * 7 : value }); }}/></div>}<h3 className={styles.priorityTitle}>{t("Select message priority")}</h3><p>{t("Set the priority of this campaign relative to other in-app message campaigns and Canvases.")}</p><div className={styles.field}><b>{t("Message priority")}</b></div><div className={styles.priorityControl}><Select label="Message priority" t={t} value={priority} options={[["High", "High priority (Shown before other messages)"], ["Medium", "Medium priority"], ["Low", "Low priority (Shown after other messages)"]].map(([value, label]) => ({ value, label: t(label) }))} onChange={iamPriority => change({ iamPriority })}/><button type="button" className={styles.priorityButton} onClick={() => setPriorityOpen(true)}>{t("View exact priority")}</button></div></section>
    {priorityOpen && <PriorityDialog metadata={metadata} draft={draft} priority={priority} t={t} close={() => setPriorityOpen(false)}/>}
  </div>;
}
