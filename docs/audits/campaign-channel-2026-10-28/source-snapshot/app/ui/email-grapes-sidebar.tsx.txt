"use client";

import { useEffect, useState, type ReactNode, type PointerEvent as ReactPointerEvent } from "react";
import type { Component, Editor } from "grapesjs";
import { AlignCenter, AlignLeft, AlignRight, ChevronDown, ChevronUp, Code2, Copy, Film, Heading1, Image as ImageIcon, LayoutGrid, List, Menu, Monitor, MousePointer2, MoveVertical, PanelTop, Pilcrow, Rows3, SeparatorHorizontal, Share2, Smartphone, Smile, Trash2, X } from "lucide-react";
import { emailFontFamilies, escapeHtml, type EmailStyle } from "@/lib/email-editor-model";
import { emailRowLayouts, grapesRowMarkup } from "@/lib/email-grapes-template";
import styles from "./email-grapes-sidebar.module.css";
import { beginPaletteDrag } from "./email-grapes-drag";

const groups = [
  { name: "BASIC BLOCKS", blocks: [["braze-title", "TITLE", Heading1], ["braze-text", "PARAGRAPH", Pilcrow], ["braze-list", "LIST", List], ["braze-button", "BUTTON", MousePointer2], ["braze-divider", "DIVIDER", SeparatorHorizontal], ["braze-spacer", "SPACER", MoveVertical]] },
  { name: "MEDIA", blocks: [["braze-image", "IMAGE", ImageIcon], ["braze-video", "VIDEO", Film], ["braze-social", "SOCIAL", Share2], ["braze-icon", "ICONS", Smile]] },
  { name: "ADVANCED", blocks: [["braze-html", "HTML", Code2], ["braze-menu", "MENU", Menu]] },
] as const;
const rowName = (widths: readonly number[]) => `${widths.join(" / ")} row`;
const inner = (component: Component) => component.getInnerHTML();
const number = (value: unknown, fallback: number) => { const parsed = Number.parseFloat(String(value ?? "")); return Number.isFinite(parsed) ? parsed : fallback; };

function Field({ label, children }: { label: string; children: ReactNode }) { return <label className={styles.field}><span>{label}</span>{children}</label>; }
function Color({ label, value, onChange }: { label: string; value: string; onChange: (value: string) => void }) { return <div className={styles.control}><span>{label}</span><div className={styles.color}><input aria-label={label} type="color" value={/^#[0-9a-f]{6}$/i.test(value) ? value : "#ffffff"} onChange={e => onChange(e.target.value)}/><input aria-label={`${label} hex`} value={value} onChange={e => onChange(e.target.value)}/></div></div>; }
function Numeric({ label, value, onChange, min = 0, max = 1000, suffix = "px" }: { label: string; value: number; onChange: (value: number) => void; min?: number; max?: number; suffix?: string }) { return <div className={styles.control}><span>{label}</span><div className={styles.numeric}><button type="button" aria-label={`Decrease ${label}`} disabled={value <= min} onClick={() => onChange(Math.max(min, value - 1))}>−</button><input aria-label={label} type="number" min={min} max={max} value={value} onChange={e => { if (e.target.value !== "") onChange(Math.max(min, Math.min(max, Number(e.target.value)))); }}/><small>{suffix}</small><button type="button" aria-label={`Increase ${label}`} disabled={value >= max} onClick={() => onChange(Math.min(max, value + 1))}>+</button></div></div>; }
function Alignment({ value, onChange }: { value: string; onChange: (value: string) => void }) { return <div className={styles.control}><span>Alignment</span><div className={styles.segments}>{([['left', AlignLeft], ['center', AlignCenter], ['right', AlignRight]] as const).map(([name, Icon]) => <button key={name} type="button" aria-label={`Align ${name}`} aria-pressed={value === name} onClick={() => onChange(name)}><Icon size={17}/></button>)}</div></div>; }
function Section({ label, children, initiallyOpen = true }: { label: string; children: ReactNode; initiallyOpen?: boolean }) { const [open, setOpen] = useState(initiallyOpen); return <section className={styles.section}><button type="button" className={styles.sectionHead} aria-expanded={open} onClick={() => setOpen(!open)}>{label}{open ? <ChevronUp size={16}/> : <ChevronDown size={16}/>}</button>{open && children}</section>; }

export default function GrapesSidebar({ editor, selected, tab, onTab, globalStyle, setGlobalStyle, onChange }: {
  editor: Editor | null; selected: { cid: string; type: string; label: string; attributes: Record<string, string> } | null;
  tab: "content" | "rows" | "settings"; onTab: (tab: "content" | "rows" | "settings") => void;
  globalStyle: EmailStyle; setGlobalStyle: (style: EmailStyle) => void; onChange: () => void;
}) {
  const [rowSource, setRowSource] = useState("empty"), [contentBlocks, setContentBlocks] = useState<Array<{ id: string; name: string; data?: { content?: string } }>>([]), [resourceError, setResourceError] = useState(""), [resourcesLoading, setResourcesLoading] = useState(false);
  const [paddingSides, setPaddingSides] = useState(false);
  useEffect(() => {
    if (rowSource !== "content") return;
    let cancelled = false; setResourcesLoading(true); setResourceError("");
    void fetch("/api/resources/content-blocks").then(r => r.ok ? r.json() : Promise.reject(new Error("Content Blocks could not be loaded."))).then(data => { if (!cancelled) setContentBlocks(data.data ?? []); }).catch(error => { if (!cancelled) setResourceError(error.message); }).finally(() => { if (!cancelled) setResourcesLoading(false); });
    return () => { cancelled = true; };
  }, [rowSource]);
  const component = selected ? editor?.getSelected() : undefined;
  const get = (key: string, fallback = "") => String(component?.getAttributes()[key] ?? fallback);
  const update = (attrs: Record<string, string | number>) => { component?.addAttributes(attrs); onChange(); };
  const content = component ? inner(component) : "";
  const setContent = (value: string) => { component?.components(value); onChange(); };
  const kind = get("css-class");
  const isRow = selected?.type === "mj-section", isColumn = selected?.type === "mj-column";
  const text = selected?.type === "mj-text", button = selected?.type === "mj-button", image = selected?.type === "mj-image", divider = selected?.type === "mj-divider", spacer = selected?.type === "mj-spacer", raw = selected?.type === "mj-raw";
  const isList = text && /<(ul|ol)\b/.test(content), isTitle = text && /<h[1-6]\b/.test(content);
  const title = isRow ? "ROW PROPERTIES" : isColumn ? "COLUMN PROPERTIES" : "CONTENT PROPERTIES";
  const append = (markup: string, row = false) => {
    if (!editor) return;
    let parent = editor.getSelected();
    while (parent && parent.get("type") !== "mj-column") parent = parent.parent();
    if (row) parent = editor.getWrapper()?.findType("mj-body")[0];
    if (!parent) {
      const body = editor.getWrapper()?.findType("mj-body")[0];
      parent = body?.findType("mj-column").at(-1);
      if (!parent && body) { const created = body.append(grapesRowMarkup([100], globalStyle.contentBg))[0]; parent = created?.findType("mj-column")[0]; parent?.components().reset(); }
    }
    const inserted = parent?.append(markup)[0];
    if (inserted) editor.select(inserted);
    onChange();
  };
  const startDrag = (id: string, event: ReactPointerEvent<HTMLButtonElement>) => {
    if (editor) beginPaletteDrag(editor, id, event.nativeEvent, onChange);
  };
  const blockButton = (id: string, label: string, Icon: typeof Heading1) => <button type="button" key={id} className={styles.block} disabled={!editor} aria-label={`Add ${label.toLowerCase()} block`} title={`Drag ${label.toLowerCase()} into your email`} onPointerDown={event => startDrag(id, event)} onClick={() => { const block = editor?.Blocks.get(id); if (block) append(String(block.getContent())); }}><Icon size={38} strokeWidth={1.8}/><span>{label}</span></button>;
  const clone = () => { if (!component) return; const parent = component.parent(); const copy = parent?.components().add(component.toJSON(), { at: parent.components().indexOf(component) + 1 }); if (copy) editor?.select(copy); onChange(); };
  const clearSelection = () => editor?.select();
  const setHide = (target: string) => { const classes = get("css-class").split(/\s+/).filter(value => value && value !== "email-hide-mobile" && value !== "email-hide-desktop"); if (target !== "none") classes.push(`email-hide-${target}`); update({ "css-class": classes.join(" ") }); };
  const padding = get("padding", "10px").split(/\s+/).map(v => number(v, 0));
  const sides = [padding[0], padding[1] ?? padding[0], padding[2] ?? padding[0], padding[3] ?? padding[1] ?? padding[0]];

  return <aside className={styles.sidebar} aria-label="Email design tools">
    <div role="tablist" aria-label="Email design panels" className={styles.tabs}>{([['content', LayoutGrid], ['rows', Rows3], ['settings', PanelTop]] as const).map(([name, Icon]) => <button type="button" key={name} role="tab" aria-selected={tab === name} onClick={() => { clearSelection(); onTab(name); }}><Icon size={19}/>{name.toUpperCase()}</button>)}</div>
    {component ? <div key={selected?.cid} className={styles.properties}>
      <div className={styles.propertyHeader}><b>{title}</b><button type="button" aria-label="Duplicate selected block" onClick={clone}><Copy size={16}/></button><button type="button" aria-label="Delete selected block" onClick={() => { component.remove(); clearSelection(); onChange(); }}><Trash2 size={16}/></button><button type="button" aria-label="Close properties" onClick={clearSelection}><X size={17}/></button></div>
      <div className={styles.selectedKind}>{isTitle ? "TITLE" : isList ? "LIST" : kind.includes("video") ? "VIDEO" : selected?.label}</div>
      {(text || button) && <>
        <Field label="Font family"><select value={get("font-family", globalStyle.fontFamily)} onChange={e => update({ "font-family": e.target.value })}>{[...new Set([get("font-family", globalStyle.fontFamily), ...emailFontFamilies])].map(font => <option key={font}>{font}</option>)}</select></Field>
        <Numeric label="Font size" value={number(get("font-size"), 15)} min={6} max={96} onChange={v => update({ "font-size": `${v}px` })}/>
        <Field label="Font weight"><select value={get("font-weight", "400")} onChange={e => update({ "font-weight": e.target.value })}>{[400, 500, 600, 700, 800, 900].map(v => <option key={v} value={v}>{v === 400 ? "Regular" : v === 700 ? "Bold" : v}</option>)}</select></Field>
        <Color label="Text color" value={get("color", button ? "#ffffff" : globalStyle.textColor)} onChange={v => update({ color: v })}/>
        <Alignment value={get("align", "left")} onChange={v => update({ align: v })}/>
        <Numeric label="Line height" value={number(get("line-height"), 1.5)} min={1} max={4} suffix="" onChange={v => update({ "line-height": v })}/>
        <Numeric label="Letter spacing" value={number(get("letter-spacing"), 0)} min={-4} max={20} onChange={v => update({ "letter-spacing": `${v}px` })}/>
        {button ? <Field label="Button text"><input value={content.replace(/<[^>]+>/g, "")} onChange={e => setContent(escapeHtml(e.target.value))}/></Field> : <Field label={isList ? "List items (one per line)" : "Text content"}><textarea value={isList ? [...content.matchAll(/<li\b[^>]*>([\s\S]*?)<\/li>/gi)].map(m => m[1].replace(/<[^>]*>/g, "")).join("\n") : content} onChange={e => { if (!isList) setContent(e.target.value); else { const tag = /<ol\b/.test(content) ? "ol" : "ul"; setContent(`<${tag}>${e.target.value.split("\n").map(line => `<li>${escapeHtml(line)}</li>`).join("")}</${tag}>`); } }}/></Field>}
        {isTitle && <Field label="Title style"><select value={content.match(/<h([1-6])\b/)?.[1] ?? "1"} onChange={e => setContent(content.replace(/<(\/?)(h)[1-6]\b/g, `<$1h${e.target.value}`))}>{[1,2,3,4,5,6].map(level => <option key={level} value={level}>Heading {level}</option>)}</select></Field>}
        {isList && <Field label="List type"><select value={/<ol\b/.test(content) ? "ol" : "ul"} onChange={e => setContent(content.replace(/<(\/?)(ul|ol)\b/g, `<$1${e.target.value}`))}><option value="ul">Unordered</option><option value="ol">Ordered</option></select></Field>}
      </>}
      {(image || button) && <Section label="ACTION"><Field label="Link type"><select value={get("href").startsWith("mailto:") ? "mailto:" : get("href").startsWith("tel:") ? "tel:" : get("href").startsWith("sms:") ? "sms:" : "url"} onChange={e => update({ href: (e.target.value === "url" ? "https://" : e.target.value) })}><option value="url">Open web page</option><option value="mailto:">Send email</option><option value="tel:">Make a call</option><option value="sms:">Send SMS</option></select></Field><Field label="URL"><input value={get("href")} onChange={e => update({ href: e.target.value })}/></Field><Field label="Target"><select value={get("target", "_blank")} onChange={e => update({ target: e.target.value })}><option value="_blank">New window</option><option value="_self">Same window</option></select></Field></Section>}
      {image && <><Field label={kind.includes("video") ? "Video thumbnail URL" : "Image URL"}><input value={get("src")} onChange={e => update({ src: e.target.value })}/></Field><Field label="Alternate text"><input value={get("alt")} onChange={e => update({ alt: e.target.value })}/></Field><Numeric label="Width" value={number(get("width"), 300)} min={1} max={1200} onChange={v => update({ width: `${v}px` })}/><Alignment value={get("align", "center")} onChange={v => update({ align: v })}/></>}
      {button && <><Color label="Background color" value={get("background-color", "#6736d8")} onChange={v => update({ "background-color": v })}/><Numeric label="Border radius" value={number(get("border-radius"), 4)} max={100} onChange={v => update({ "border-radius": `${v}px` })}/><Field label="Inner padding"><input value={get("inner-padding", "10px 25px")} onChange={e => update({ "inner-padding": e.target.value })}/></Field><Field label="Border"><input value={get("border", "none")} onChange={e => update({ border: e.target.value })}/></Field></>}
      {divider && <><Color label="Line color" value={get("border-color", "#dcd9e0")} onChange={v => update({ "border-color": v })}/><Numeric label="Line weight" value={number(get("border-width"), 1)} max={30} onChange={v => update({ "border-width": `${v}px` })}/><Field label="Line style"><select value={get("border-style", "solid")} onChange={e => update({ "border-style": e.target.value })}>{["solid", "dashed", "dotted"].map(v => <option key={v}>{v}</option>)}</select></Field><Numeric label="Width" value={number(get("width"), 100)} min={5} max={100} suffix="%" onChange={v => update({ width: `${v}%` })}/><Alignment value={get("align", "center")} onChange={v => update({ align: v })}/></>}
      {spacer && <Numeric label="Height" value={number(get("height"), 60)} max={500} onChange={v => update({ height: `${v}px` })}/>}
      {raw && <Field label="HTML code"><textarea className={styles.code} value={content} onChange={e => setContent(e.target.value)}/></Field>}
      {(selected?.type === "mj-social" || selected?.type === "mj-navbar") && <Section label={selected.type === "mj-social" ? "SOCIAL LINKS" : "MENU ITEMS"}>{component.components().map((child: Component, index: number) => <div key={child.cid} className={styles.item}><Field label={selected.type === "mj-social" ? "Network" : "Label"}><input value={selected.type === "mj-social" ? String(child.getAttributes().name ?? "") : inner(child)} onChange={e => { if (selected.type === "mj-social") child.addAttributes({ name: e.target.value }); else child.components(escapeHtml(e.target.value)); onChange(); }}/></Field><Field label="Link URL"><input value={String(child.getAttributes().href ?? "")} onChange={e => { child.addAttributes({ href: e.target.value }); onChange(); }}/></Field><button type="button" className={styles.textButton} onClick={() => { child.remove(); onChange(); }}>Remove item {index + 1}</button></div>)}<button type="button" className={styles.addItem} onClick={() => { component.append(selected.type === "mj-social" ? '<mj-social-element name="facebook" href="https://"></mj-social-element>' : '<mj-navbar-link href="https://">Menu item</mj-navbar-link>'); onChange(); }}>+ Add item</button></Section>}
      {(isRow || isColumn) && <>
        <Color label={isRow ? "Row background color" : "Column background color"} value={get("background-color", "#ffffff")} onChange={v => update({ "background-color": v })}/>
        {isRow && <><Field label="Background image URL"><input value={get("background-url")} onChange={e => update({ "background-url": e.target.value })}/></Field><Field label="Background image size"><select value={get("background-size", "auto")} onChange={e => update({ "background-size": e.target.value })}>{["auto", "cover", "contain"].map(value => <option key={value}>{value}</option>)}</select></Field><Field label="Background position"><select value={get("background-position", "center top")} onChange={e => update({ "background-position": e.target.value })}>{["center top", "center center", "center bottom", "left top", "right top"].map(value => <option key={value}>{value}</option>)}</select></Field><Numeric label="Border radius" value={number(get("border-radius"), 0)} max={100} onChange={v => update({ "border-radius": `${v}px` })}/><Section label="CUSTOMIZE COLUMNS">{component.findType("mj-column").map((column, index, columns) => <div key={column.cid} className={styles.column}><Numeric label={`Column ${index + 1} width`} value={number(column.getAttributes().width, 100 / columns.length)} min={1} max={99} suffix="%" onChange={v => { if (columns.length === 1) return; const others = columns.filter(c => c !== column); const total = others.reduce((sum, c) => sum + number(c.getAttributes().width, 100 / columns.length), 0); column.addAttributes({ width: `${v}%` }); let remaining = 100 - v; others.forEach((c, i) => { const width = i === others.length - 1 ? remaining : Math.round((100 - v) * number(c.getAttributes().width, 100 / columns.length) / total * 100) / 100; c.addAttributes({ width: `${width}%` }); remaining -= width; }); onChange(); }}/><button type="button" className={styles.textButton} onClick={() => editor?.select(column)}>Edit column {index + 1}</button></div>)}</Section></>}
      </>}
      <Section label="BLOCK OPTIONS"><Color label="Content background color" value={get("container-background-color", "#ffffff")} onChange={v => update({ "container-background-color": v })}/><div className={styles.control}><span>Padding</span><label className={styles.check}><input type="checkbox" checked={paddingSides} onChange={e => setPaddingSides(e.target.checked)}/> More options</label></div>{paddingSides ? ["Top", "Right", "Bottom", "Left"].map((side, index) => <Numeric key={side} label={`${side} padding`} value={sides[index]} max={100} onChange={v => { const next = [...sides]; next[index] = v; update({ padding: next.map(n => `${n}px`).join(" ") }); }}/>) : <Numeric label="All sides" value={sides[0]} max={100} onChange={v => update({ padding: `${v}px` })}/>}<div className={styles.control}><span>Hide on</span><div className={styles.segments}>{([['none', null], ['desktop', Monitor], ['mobile', Smartphone]] as const).map(([value, Icon]) => <button type="button" key={value} aria-label={`Hide on ${value}`} aria-pressed={value === "none" ? !/email-hide-(mobile|desktop)/.test(kind) : kind.includes(`email-hide-${value}`)} onClick={() => setHide(value)}>{Icon ? <Icon size={17}/> : "None"}</button>)}</div></div></Section>
    </div> : tab === "content" ? <div className={styles.catalog}>{groups.map((group, index) => <Section key={group.name} label={group.name} initiallyOpen={index === 0}><div className={styles.blocks}>{group.blocks.map(([id, label, Icon]) => blockButton(id, label, Icon))}</div></Section>)}</div> : tab === "rows" ? <div className={styles.rows}><select aria-label="Row library" value={rowSource} onChange={e => setRowSource(e.target.value)}><option value="empty">Empty rows</option><option value="content">Content Blocks</option></select>{rowSource === "empty" ? emailRowLayouts.map((widths, index) => <button type="button" key={widths.join("-")} className={styles.rowCard} aria-label={`Add ${rowName(widths)}`} disabled={!editor} onPointerDown={event => startDrag(`braze-row-${index}`, event)} onClick={() => append(grapesRowMarkup(widths, globalStyle.contentBg), true)}><span>{widths.map((width, i) => <i key={i} style={{ flex: width }}><em/></i>)}</span></button>) : <div className={styles.savedRows}>{resourcesLoading ? <p role="status">Loading Content Blocks…</p> : resourceError ? <p role="alert">{resourceError}</p> : contentBlocks.length ? contentBlocks.map(block => <button type="button" key={block.id} disabled={!editor} onClick={() => append(`<mj-section><mj-column><mj-text>{{content_blocks.\${${block.name}}}}</mj-text></mj-column></mj-section>`, true)}><Code2 size={20}/>{block.name}</button>) : <p>No Content Blocks have been saved in this workspace.</p>}</div>}</div> : <div className={styles.settings}>
      <Numeric label="Content area width" value={globalStyle.contentWidth} min={320} max={800} onChange={v => setGlobalStyle({ ...globalStyle, contentWidth: v })}/><input className={styles.range} aria-label="Content area width slider" type="range" min={320} max={800} step={10} value={globalStyle.contentWidth} onChange={e => setGlobalStyle({ ...globalStyle, contentWidth: Number(e.target.value) })}/>
      <Color label="Background color" value={globalStyle.canvasBg} onChange={v => setGlobalStyle({ ...globalStyle, canvasBg: v })}/><Color label="Content area background color" value={globalStyle.contentBg} onChange={v => setGlobalStyle({ ...globalStyle, contentBg: v })}/><Field label="Default font"><select value={globalStyle.fontFamily} onChange={e => setGlobalStyle({ ...globalStyle, fontFamily: e.target.value })}>{[...new Set([globalStyle.fontFamily, ...emailFontFamilies])].map(font => <option key={font}>{font}</option>)}</select></Field><Color label="Text color" value={globalStyle.textColor} onChange={v => setGlobalStyle({ ...globalStyle, textColor: v })}/><Color label="Link color" value={globalStyle.linkColor} onChange={v => setGlobalStyle({ ...globalStyle, linkColor: v })}/>
    </div>}
  </aside>;
}
